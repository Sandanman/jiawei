## 探测代码片段（推断规范示例）

```typescript
import { Request, Response } from 'express'
import { asyncHandler } from '../utils/asyncHandler'

/**
 * POST /api/xxx/action
 * Body: { field: string }
 */
export const actionHandler = asyncHandler(async (req: Request, res: Response) => {
    const { field } = req.body || {}

    if (!field) {
        return res.fail('参数不能为空')
    }

    try {
        const result = await someService(field)
        res.success(result, '操作成功')
    } catch (err) {
        res.fail((err as Error).message)
    }
})
```
